using System.Collections;
using System.Collections.Generic;
using System.Linq;
using UnityEngine;

public class BattleManager : MonoBehaviour//战斗管理器
{
    public Card_SO card_SO;
    void OnEnable()
    {
        EventHandler.Execute += Battle;
    }
    void OnDisable()
    {
        EventHandler.Execute -= Battle;
    }
    public void Battle(GameObject Caster,GameObject Target, int ID)
    {
        if (Target == null)
            return;
        //传入目标和功能函数
        //检测自身状态
        //执行卡牌功能
        CardData Date = card_SO.cardDates.FirstOrDefault(i => i.ID == ID);
        if (Date != null)
        {
            Date.function.value = Date.date;
            //对特效进行赋值
            Caster.GetComponent<VfxPlayer>().SetVfxAnimator(Date.animator);
            Target.GetComponent<VfxPlayer>().SetVfxAnimator(Date.animator);
            //再到卡牌功能类里启动
            Date.function.Functions(Caster,Target);
        }
    }
}

using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;
public class Energy : MonoBehaviour
{
    public int CurrEnergy;//当前能量
    public int MaxEnergy;//能量上限
    public Text text;//能量UI
    public GameObject game;
    public static Energy instance;//单例模式
    void Awake()
    {
        if (instance != null)
        {
            Destroy(gameObject);
            return;
        }
        else
        {
            instance = this;
        }
        InitEnergy();
    }
    void SetEnergy()
    {
        text.text = CurrEnergy.ToString();
    }
    public void InitEnergy()
    {
        text.text = MaxEnergy.ToString();
        CurrEnergy = MaxEnergy;
    }
    public bool ConsumEnergy(int data)
    {
        if (CurrEnergy - data < 0)
        {
            StartCoroutine(NotEnergy());
            return false;
        }
        CurrEnergy -= data;
        SetEnergy();
        return true;
    }
    IEnumerator NotEnergy()
    {
        game.SetActive(true);
        yield return new WaitForSeconds(0.2f);
        game.SetActive(false);
    }
}

using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;
public class Health : MonoBehaviour
{
    public float MaxHealth;//最大生命值
    public float CurrHealth;//当前生命值
    public Slider slider;
    public Text text;//血量显示
    private Animator animator;
    private Image image;//图片
    void Awake()
    {
        image=GetComponent<Image>();
        animator = GetComponent<Animator>();
        StartHealth(MaxHealth);
    }
    public void StartHealth(float health)
    {
        MaxHealth = CurrHealth = health;
        text.text = CurrHealth.ToString() + "/" + MaxHealth.ToString();
        slider.value = 1;
    }
    public void SetCurrHealth(float data)
    {
        if(CurrHealth+data>=MaxHealth)
        {
            CurrHealth=MaxHealth;
        }
        else
        {
            CurrHealth+=data;
        }
    }
    public void SetMaxHealth(float data)
    {
        MaxHealth+=data;
    }
    public void SetHealth()
    {
        text.text = CurrHealth.ToString() + "/" + MaxHealth.ToString();
        if( slider.value > CurrHealth / MaxHealth)
        {
            StartCoroutine(IHurt());
        }
        slider.value = CurrHealth / MaxHealth;
        
    }
    IEnumerator IHurt()//受伤
    {
        if (gameObject.tag == "NPC")
        {
            EnemyManager.instance.EnemyDeath = false;
            Debug.Log("怪物受伤1");
        }
        animator.SetTrigger("受伤");
        yield return null;
        Color c=image.color;
        
        AnimatorStateInfo stateInfo = animator.GetCurrentAnimatorStateInfo(0);
        yield return new WaitForSeconds(stateInfo.length/5);
        c=Color.red;
        image.color=c;
        yield return new WaitForSeconds(stateInfo.length/5);
        c=Color.white;
        image.color=c;
        yield return new WaitForSeconds(stateInfo.length/5);
        c=Color.red;
        image.color=c;
        yield return new WaitForSeconds(stateInfo.length/5);
        c=Color.white;
        image.color=c;
        yield return new WaitForSeconds(stateInfo.length/5);
        if (CurrHealth <= 0)
        {
            StartCoroutine(IDeath());
        }
        else
        {
            if (gameObject.tag == "NPC")
            {
                EnemyManager.instance.EnemyDeath = true;
                Debug.Log("怪物受伤2");
            }
        }
        c=Color.white;
        image.color=c;
    }
    IEnumerator IDeath()//死亡
    {
        animator.SetTrigger("死亡");
        //通知怪物管理器回收
        if (gameObject.tag == "NPC")
        {
            EnemyManager.instance.EnemyDeath = false;
            AnimatorStateInfo stateInfo = animator.GetCurrentAnimatorStateInfo(0);
            yield return new WaitForSeconds(stateInfo.length + 0.5f);
            EventHandler.CallEnemyDeath(gameObject);
        }
        else
        {
            //玩家死亡
            EventHandler.CallLose();
            Debug.Log("玩家死亡，游戏结算");
            //EventHandler.CallLose();
        }
        Debug.Log("死亡");
        yield return null;
    }
}

using System.Collections;
using System.Collections.Generic;
using UnityEngine;
public enum TurnState
{
    PlayerTurn,//玩家回合
    EnemyTurn//敌人回合
}
public class TurnManager : MonoBehaviour//回合控制器
{
    public GameObject game;
    void OnEnable()
    {
        EventHandler.PlayerTrun+=OnPlayerTrun;
        EventHandler.EnemyTurn+=OnEnemyTurn;
    }
    void OnDisable()
    {
        EventHandler.PlayerTrun-=OnPlayerTrun;
        EventHandler.EnemyTurn-=OnEnemyTurn;
    }
    public void OnPlayerTrun()//玩家回合
    {
        //发牌
        EventHandler.CallDeal();
        //开启回合结束
        game.SetActive(true);
        Energy.instance.InitEnergy();
        //执行一些玩家操作
    }
    public void OnEnemyTurn()//敌人回合
    {
        //关闭回合结束
        game.SetActive(false);
        //回收玩家的卡牌
        EventHandler.CallRecycleAllCard();
        //敌人攻击
        //通知敌人攻击
        EventHandler.CallEnemyAttack();
        StartCoroutine(EnemyTurn());
        Debug.Log("敌人回合");
    }
    private IEnumerator EnemyTurn()
    {
        yield return new WaitUntil(()=>EnemyManager.instance.PlayerTrun);//等待敌人攻击完成
        EnemyManager.instance .PlayerTrun=false;
        Debug.Log("敌人攻击");
        //进入玩家回合
        OnPlayerTrun();
    }
}

        {
            sfxSource.volume = sfx;
            audioData.sfxSource = sfx;
        }

    }

    {
        sfxSource.Play();
    }

using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;
using UnityEngine.EventSystems;
using DG.Tweening;

public class Card : MonoBehaviour, IBeginDragHandler, IDragHandler, IEndDragHandler, IPointerEnterHandler, IPointerExitHandler
{
    public int ID;//卡牌ID
    public GameObject player;//玩家
    public int energy;
    private Vector3 pos;
    private CanvasGroup canvasGroup;
    private Image image;//卡牌图片
    public string Description;//卡牌描述
    private TextUI textUI;//卡牌描述
    private Intent intent;//卡牌描述显示的位置
    private int index;//位置
    private Vector3 Angle;
    void Awake()
    {
        intent = GetComponent<Intent>();
        image = GetComponent<Image>();
        canvasGroup = GetComponent<CanvasGroup>();
        textUI = GetComponent<TextUI>();
    }
    public void SetImage(Sprite sprite)
    {
        image.sprite = sprite;
    }
    public void OnBeginDrag(PointerEventData eventData)
    {
        pos = (Vector3)eventData.position - transform.position;
        transform.localScale = new Vector3(1.2f, 1.2f, 0);
        canvasGroup.blocksRaycasts = false;
        transform.DORotate(Vector3.zero, 0.5f);
    }
    public void OnDrag(PointerEventData eventData)
    {
        transform.position = Input.mousePosition - pos;
        transform.DORotate(Vector3.zero, 0.5f);
    }
    public void OnEndDrag(PointerEventData eventData)
    {
        GameObject obj = eventData.pointerCurrentRaycast.gameObject;
        if (obj == null)
        {
            EventHandler.CallCardPositionUpdate();
            canvasGroup.blocksRaycasts = true;
            return;
        }
        if (obj.tag == "NPC")
        {
            if (Energy.instance.ConsumEnergy(energy))
            {
                //执行卡牌功能
                Debug.Log("执行卡牌");
                //检测卡牌是否能执行
                EventHandler.CallExecute(player, obj, ID);
                //执行完卡牌将数据给到弃牌组
                EventHandler.CallRecycleCard(gameObject, ID);
                //播放玩家攻击动画
                //播放音效
                EventHandler.CallAudiosfx();
            }
            else
            {
                Debug.Log("没能量了");
            }
        }
        else
        {
            //回到原位置
            Debug.Log("回到原位置");
            //通知
        }
        transform.DORotate(Angle, 0.5f);
        transform.localScale = new Vector3(1f, 1f, 0);
        EventHandler.CallCardPositionUpdate();
        Debug.Log(obj.tag);
        canvasGroup.blocksRaycasts = true;
    }

    public void OnPointerEnter(PointerEventData eventData)
    {
        transform.localScale = new Vector3(1.2f, 1.2f, 0);
        index = transform.GetSiblingIndex();
        transform.SetAsLastSibling();
        intent.SetDistance();
        textUI.OnShow(Description);
        Angle = transform.localEulerAngles;
        transform.DORotate(Vector3.zero, 0.5f);
        //transform.position = new Vector3();
        //使用计时器
        //通知显示卡牌效果文字
        //EventHandler.CallShow(Description);
    }
    public void OnPointerExit(PointerEventData eventData)
    {
        transform.localScale = new Vector3(1f, 1f, 0);
        transform.SetSiblingIndex(index);
        textUI.OnHide();
        transform.DORotate(Angle, 0.5f);
        //关闭卡牌效果文字
        //EventHandler.CallHide();
    }
}

using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.Pool;
using System.Linq;
using DG.Tweening;
public class CardManager : MonoBehaviour
{
    [Header("卡牌预制体")] public GameObject Prefabe;//卡牌预制体
    [Header("最大卡牌数量")] public int MaxCard;//最大卡牌数量
    [Header("每次发牌数量")] public int DrawCount;
    [Header("玩家")] public GameObject player;
    [Header("扇形布局")]
    //距离圆心距离
    public float W = 300f;                 // 半宽，按你手牌区调
    public float thetaDeg = 30f;
    public float CAngle;//每张牌的固定角度
    public Card_SO card_SO;//卡牌SO文件 用于存储所有卡牌数据 
    //当前手牌
    public List<int> Hand;
    private IObjectPool<GameObject> CardPool;//对象池
    //发牌组
    private List<int> DrawPile;
    //弃牌组
    private List<int> DiscardPile;

    private float width;
    private float height;
    private RectTransform rect;
    void Awake()
    {
        //初始化List
        DrawPile = new List<int>();
        DiscardPile = new List<int>();
        //初始化对象池
        CardPool = new ObjectPool<GameObject>(createFunc: CreatePooledItem,
            actionOnGet: OnTakeFromPool,
            actionOnRelease: OnReturnedToPool,
            actionOnDestroy: OnDestroyPoolObject);
        //获取预制体宽高
        RectTransform rt = Prefabe.GetComponent<RectTransform>();
        width = rt.rect.width;
        height = rt.rect.height;
        rect = GetComponent<RectTransform>();
    }
    void Start()
    {
        DrawPile.AddRange(Hand);//第一次将所有手牌加到发牌组
                                // StartDeal();
    }
    void OnEnable()
    {
        EventHandler.CardPositionUpdate += UpdateHandLayout;
        EventHandler.RecycleCard += OnRecycleCard;
        EventHandler.Deal += StartDeal;
        EventHandler.RecycleAllCard += OnRecycleAllCard;
        EventHandler.CardID += OnCardID;
    }
    void OnDisable()
    {
        EventHandler.CardPositionUpdate -= UpdateHandLayout;
        EventHandler.RecycleCard -= OnRecycleCard;
        EventHandler.Deal -= StartDeal;
        EventHandler.RecycleAllCard -= OnRecycleAllCard;
        EventHandler.CardID -= OnCardID;
    }

    private void OnCardID(int ID)
    {
        //将新加入的卡牌加入发牌组，同时更新手牌
        DrawPile.Add(ID);
        Hand.Add(ID);
    }

    private void StartDeal()//开始发牌
    {
        StartCoroutine(Deal());
    }
    /// <summary>
    /// 回收卡牌
    /// </summary>
    /// <param name="obj">回收的物体</param>
    /// <param name="ID">回收的ID</param>
    private void OnRecycleCard(GameObject obj, int ID)
    {
        DiscardPile.Add(ID);
        CardPool.Release(obj);
        //UpdateHandLayout();
    }
    #region 对象池
    // 1. 当池中没有对象时，如何创建新对象
    private GameObject CreatePooledItem()
    {
        GameObject obj = Instantiate(Prefabe, transform);
        obj.transform.position=Vector3.zero;
        return obj;
    }
    // 2. 从池中获取对象时调用
    private void OnTakeFromPool(GameObject obj)
    {
        obj.SetActive(true);
    }
    // 3. 将对象归还给池时调用
    private void OnReturnedToPool(GameObject obj)
    {
        obj.transform.position=Vector3.zero;
        obj.SetActive(false);
    }
    // 4. 当池满销毁多余对象时调用
    private void OnDestroyPoolObject(GameObject obj)
    {
        Destroy(obj);
    }
    #endregion
    #region 修改卡牌位置
    private List<Card> cards = new List<Card>();//自身所有的子物体卡牌
    private void GetChildrenCard()
    {
        cards.Clear();
        foreach (Transform child in transform)
        {
            if (child.gameObject.activeSelf)
            {
                cards.Add(child.GetComponent<Card>());
            }
        }
    }
    //使用算法计算卡牌的位置
    public void UpdateHandLayout()
    {
        GetChildrenCard();
        int count = cards.Count;
        if (count == 0) return;
        float thetaRad = thetaDeg * Mathf.Deg2Rad;
        float R = W / Mathf.Sin(thetaRad);
        //每一步角度
        float Angle = (thetaDeg - (-thetaDeg)) / (count - 1);
        float CardAngle = -thetaDeg;
        float rand = CardAngle * Mathf.PI / 180f; ;
        float data = R * Mathf.Cos(rand);
        if(CAngle*cards.Count<thetaDeg*2)
        {
            Angle=CAngle;
            CardAngle=-CAngle*cards.Count/2;
        }
        for (int i = 0; i < count; i++)
        {
            RectTransform cardRect = cards[i].GetComponent<RectTransform>();
            rand = CardAngle * Mathf.PI / 180f;
            float X = R * Mathf.Sin(rand);
            float Y = R * Mathf.Cos(rand);
            cardRect.DOMove(new Vector2(transform.position.x+X, Y - data),1);
           // = new Vector2(X, Y - data);
            cardRect.DORotate(new Vector3(0,0,-CardAngle), 1f);
            CardAngle += Angle;
        }
    }

    #endregion
    //使用对象池进行创建卡牌
    #region 洗牌/发牌/回收所有卡牌
    private void OnRecycleAllCard()
    {
        GetChildrenCard();
        foreach (var i in cards)
        {
            GameObject obj = i.gameObject;
            int ID = i.ID;
            OnRecycleCard(obj, ID);
        }
    }
    //使用洗牌算法
    private void Shuffle(List<int> DrawPile)
    {
        int Len = DrawPile.Count - 1;
        if (Len < DrawCount)//当卡牌小于发牌数量
        {
            DrawPile.AddRange(DiscardPile);
            Len = DrawPile.Count - 1;
            DiscardPile.Clear();//清空弃牌组
            //将弃牌组的卡牌拿过来
        }
        int i;
        for (int j = Len; j > 0; j--)
        {
            i = Random.Range(0, j + 1);
            int m = DrawPile[i];
            DrawPile[i] = DrawPile[j];
            DrawPile[j] = m;
        }
    }
    public IEnumerator Deal()//发牌
    {
        for (int i = 0; i < DrawCount; i++)
        {
            yield return new WaitForSeconds(0.2f);
            yield return StartCoroutine(Draw());
        }
    }
    private IEnumerator Draw()//发牌
    {
        if (DrawPile.Count >= 1)
        {
            //限制一下手牌数量不能超过最大值

            GameObject obj = CardPool.Get();
            if (obj != null)
            {
                Card card = obj.GetComponent<Card>();
                card.ID = DrawPile[0];//赋值ID
                //设置卡牌图片
                CardData date = card_SO.cardDates.FirstOrDefault(i => i.ID == card.ID);
                //date.function.value=date.date;
                if (date == null)
                {
                    CardPool.Release(obj);
                    yield break;
                }
                card.Description = date.Description;
                card.player = player;
                card.SetImage(date.sprite);
                card.energy = date.energy;
                UpdateHandLayout();//更新卡牌位置
                                   //修改card的数据用于初始化
                DrawPile.RemoveAt(0);
            }
        }
        else
        {
            Shuffle(DrawPile);
        }
        yield return null;
    }
    #endregion
}

    {
        if (instance != null)
        {
            Destroy(gameObject);
            return;
        }
        else if (instance == null)
        {
            instance = this;
        }
        Enemies = new List<Enemy>();
        EnemyPool = new ObjectPool<GameObject>(createFunc: CreatePooledItem,
            actionOnGet: OnTakeFromPool,
            actionOnRelease: OnReturnedToPool,

        EventHandler.EnemyAttack += EnemyAttack;

    {
        // EnemyDeath=false;
        EnemyPool.Release(obj);

            if (enemy != null)
            {
                EnemyDate date = enemy_SO.enemyDates.FirstOrDefault(i => i.ID == EnemyID);

                if (date != null)

                    enemy.SetEnemy(Player, date);
                }
            }
        }
    }

    {
        GameObject obj = Instantiate(Prefabe, transform);
        return obj;
    }
    private void OnTakeFromPool(GameObject obj)//获取敌人
    {
        obj.SetActive(true);
    }
    private void OnReturnedToPool(GameObject obj)//回收敌人
    {
        obj.SetActive(false);
        CheckWin();
    }
    private void OnDestroyPoolObject(GameObject obj)//清除多余敌人
    {
        Destroy(obj);
    }

                Enemies.Add(enemy);

using System.Collections;

public class VictoryResult : MonoBehaviour
{
    public Card_SO card_SO;
    public CardResult card1;
    public CardResult card2;
    public CardResult card3;
    void OnEnable()
    {
        EventHandler.CloseCardResult += CloseCard;
    }
    void OnDisable()
    {
        EventHandler.CloseCardResult -= CloseCard;
    }
    public void SetCard()
    {
        OpenCard();
        CardData cardData = card_SO.cardDates[Random.Range(0, card_SO.cardDates.Count)];
        card1.Description = cardData.Description;
        card1.SetImage(cardData.sprite);
        card1.ID = cardData.ID;

        cardData = card_SO.cardDates[Random.Range(0, card_SO.cardDates.Count)];
        card2.Description = cardData.Description;
        card2.SetImage(cardData.sprite);
        card2.ID = cardData.ID;

        cardData = card_SO.cardDates[Random.Range(0, card_SO.cardDates.Count)];
        card3.Description = cardData.Description;
        card3.SetImage(cardData.sprite);
        card3.ID = cardData.ID;
    }
    public void CloseCard()
    {
        card1.gameObject.SetActive(false);
        card2.gameObject.SetActive(false);
        card3.gameObject.SetActive(false);
    }
    public void OpenCard()
    {
        card1.gameObject.SetActive(true);
        card2.gameObject.SetActive(true);
        card3.gameObject.SetActive(true);
    }
}

using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.SceneManagement;
public class SceneSwitchSystem : MonoBehaviour
{
    void Awake()
    {
        DontDestroyOnLoad(gameObject);
    }
    void OnEnable()
    {
        EventHandler.Scene += SceneSwitch;
    }
    void OnDisable()
    {
        EventHandler.Scene -= SceneSwitch;
    }
    public void SceneSwitch(string name)
    {
        if (name != string.Empty)
        {
            SceneManager.LoadScene(name);
        }
    }
}
